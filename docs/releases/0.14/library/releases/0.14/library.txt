/** Ultimate bearing pressure combining N factors with shape, depth and inclination factors. Inputs: coh, phi (deg), q surcharge, gam unit weight, B, L, D footing width/length/depth, H, V loads, author (terzaghi, prandtl, meyerhof, hansen, vesic, ec7). */
bearing_capacity = LAMBDA(coh, phi, q, gam, B, L, D, H, V, author, [A_],
    LET(
        BL, B/L,
        DB, D/B,
        load, VSTACK(coh, q, gam*B/2),
        shape, bearing_shapeFactors(BL, phi, author),
        depth, bearing_depthFactors(DB, phi, author),
        inclination, bearing_inclinationFactors(H, V, BL, phi, coh, author, ),
        factors, bearing_factors(phi, author),
        SUMPRODUCT(shape, depth, inclination, factors, load)
    ));

/** Depth factors {dc; dq; dγ} for the selected bearing method. Inputs: DB = D/B, phi (deg), author. */
bearing_depthFactors = LAMBDA(DB, phi, author,
    LET(
        authors, {"terzaghi", "prandtl", "meyerhof", "hansen", "vesic", "ec7"},
        f, RADIANS(phi),
        a, IFERROR(XMATCH(LOWER(author), authors, 1), 0),
        dq, SWITCH(a, 3, IF(phi<10, 1, 1+0.1*DB*TAN(PI()/4+f/2)), 4, 1+2*IF(DB>1, ATAN(DB), DB)*TAN(f)*(1-SIN(f))^2, 5, 1+2*IF(DB>1, ATAN(DB), DB)*TAN(f)*(1-SIN(f))^2, 1),
        dc, SWITCH(a, 3, 1+0.2*DB*TAN(PI()/4+f/2), 4, 1+0.4*IF(DB>1, ATAN(DB), DB), 5, 1+0.4*IF(DB>1, ATAN(DB), DB), 1),
        dg, SWITCH(a, 3, dq, 1),
        VSTACK(dc, dq, dg)
    ));

/** Bearing capacity factors {Nc; Nq; Nγ} for the selected method. Inputs: phi (deg), author_. */
bearing_factors = LAMBDA(phi, author_,
    LET(
        authors, {"terzaghi", "prandtl", "meyerhof", "hansen", "vesic", "ec7"},
        f, RADIANS(phi),
        a, IFERROR(XMATCH(LOWER(author_), authors, 1), 0),
        Nq, IFS(a=1, EXP((3*PI()/2-f)*TAN(f))/2/COS(PI()/4+f/2)^2, a>1, EXP(PI()*TAN(f))*TAN(PI()/4+f/2)^2, TRUE, 0),
        Nc, IFS(AND(a=1, phi=0), 1+3*PI()/2, AND(a>1, phi=0), 2+PI(), TRUE, (Nq-1)/TAN(f)),
        Ng, SWITCH(a, 1, 2*(Nq+1)*TAN(f)/(1+0.4*SIN(f)), 2, 0, 3, (Nq-1)*TAN(1.4*f), 4, 1.5*(Nq-1)*TAN(f), 5, 2*(Nq+1)*TAN(f), 6, 2*(Nq-1)*TAN(f), 0, "Error"),
        VSTACK(Nc, Nq, Ng)
    ));

/** Load inclination factors {ic; iq; iγ}. Inputs: H, V horizontal/vertical loads, BL = B/L, phi (deg), coh, author, optional A_ area (default 1). */
bearing_inclinationFactors = LAMBDA(H, V, BL, phi, coh, author, [A_],
    LET(
        authors, {"terzaghi", "prandtl", "meyerhof", "hansen", "vesic", "ec7"},
        f, RADIANS(phi),
        auth, IFERROR(XMATCH(LOWER(author), authors, 1), 0),
        ach, coh*IF(ISOMITTED(A_), 1, A_),
        ac, IF(ach>H, H, ach),
        t, IF(V=0, 0, ATAN(H/V)),
        m, (2+BL)/(1+BL),
        tt, ABS(H/(V+ac)),
        Ns, bearing_factors(phi, author),
        Nc, INDEX(Ns, 1),
        Nq, INDEX(Ns, 2),
        iq, SWITCH(auth, 3, IF(phi=0, 0, (1-IF(t/f>1, 1, t/f))^2), 4, ABS(1-tt/2)^3, 5, ABS(1-tt)^m, 6, 1-t, 1),
        ig, SWITCH(auth, 3, IF(phi=0, 0, (1-IF(t/f>1, 1, t/f))^2), 4, ABS(1-0.7*tt)^3, 5, ABS(1-tt)^m, 6, IF(V=0, 1, 1-H/V), 1),
        ic, SWITCH(auth, 3, iq, 4, IF(phi=0, 0.5-SQRT(1-H/ac), (iq*Nq-1)/(Nq-1)), 5, IF(phi=0, 1-H*m/ac/Nc, (iq*Nq-1)/(Nq-1)), 6, IF(phi=0, 0.5+0.5*SQRT(1-H/ac), (iq*Nq-1)/(Nq-1)), 1),
        VSTACK(IF(ic<0, 0, ic), IF(iq<0, 0, iq), IF(ig<0, 0, ig))
    ));

/** Bearing factor Nq for the selected method (row 2 of bearing_factors). Inputs: Phi (deg), author_. */
bearing_Nq = LAMBDA(Phi, author_, INDEX(bearing_factors(Phi, author_), 2));

/** Shape factors {sc; sq; sγ} for the selected method. Inputs: BL = B/L, phi (deg), author, opt_ (any value selects the alternative Terzaghi sγ). */
bearing_shapeFactors = LAMBDA(BL, phi, author, [opt_],
    LET(
        opt, ISOMITTED(opt_),
        authors, {"terzaghi", "prandtl", "meyerhof", "hansen", "vesic", "ec7"},
        f, RADIANS(phi),
        auth, IFERROR(XMATCH(LOWER(author), authors, 1), 0),
        Ns, bearing_factors(phi, author),
        Nc, INDEX(Ns, 1),
        Nq, INDEX(Ns, 2),
        Kp, TAN(PI()/4+f/2)^2,
        sq, SWITCH(auth, 3, IF(phi<0, 1, 1+0.1*BL*Kp), 4, 1+BL*SIN(f), 5, 1+BL*TAN(f), 1),
        sc, SWITCH(auth, 1, 1+0.3*BL, 3, 1+0.2*BL*Kp, 4, IF(phi=0, 0.2*BL, 1+BL*Nq/Nc), 5, IF(phi=0, 0.2*BL, 1+BL*Nq/Nc), 1),
        sg, SWITCH(auth, 1, IF(NOT(opt), 0.6, 1-0.2*BL), 3, sq, 4, MAX(0.6, 1-0.4*BL), 5, MAX(0.6, 1-0.4*BL), 1),
        VSTACK(sc, sq, sg)
    ));

/** Skempton Nc for undrained bearing (polynomial fit). Inputs: DB = D/B (capped at 6), BL = B/L. */
bearing_Skempton = LAMBDA(DB, BL,
    LET(
        x, IF(DB>6, 6, DB),
        Nc1_, -0.0027*x^4+0.0607*x^3-0.5116*x^2+1.9247*x+6.2304,
        Nc0_, -0.0021*x^4+0.0504*x^3-0.4354*x^2+1.6433*x+PI()+2,
        Nc0_+(Nc1_-Nc0_)*BL
    ));

/** USCS symbol with ASTM group name from a two-column lookup table (symbol | descriptions). Inputs: list (normally ASTMtable; "" also uses ASTMtable), % passing No.4 and No.200, optional Cu, Cc, LL, PL, organic. */
classify_ASTM = LAMBDA(list, pf4_, pf200_, [Cu_], [Cc_], [LL_], [PL_], [organic_],
    LET(
        organic, IF(ISOMITTED(organic_), 0, organic_),
        Cu, IF(ISOMITTED(Cu_), 0, Cu_),
        Cc, IF(ISOMITTED(Cc_), 0, Cc_),
        LL, IF(ISOMITTED(LL_), 0, LL_),
        PL, IF(ISOMITTED(PL_), 0, PL_),
        sand, pf4_-pf200_,
        gravel, 100-pf4_,
        plus, 100-pf200_,
        USC_, classify_USC(pf4_, pf200_, Cu, Cc, LL, PL, organic),
        sym, IF(USC_="CL & ML", "CL-ML", USC_),
        isOrg, AND(organic>0, OR(sym="ML", sym="CL", sym="CL-ML", sym="MH", sym="CH")),
        key, IF(isOrg, IF(OR(sym="MH", sym="CH"), "OCH", "OML"), sym),
        offset, IF(AND(isOrg, OR(sym="CL", sym="CL-ML", sym="MH")), 7, 0),
        i, IFS(plus<15, 1, plus<30, IF(sand>=gravel, 2, 3), sand>=gravel, IF(gravel>=15, 5, 4), TRUE, IF(sand>=15, 7, 6)),
        other, IF(LEFT(sym, 1)="G", sand, gravel),
        j, IF(key="Pt", 1, IF(pf200_>=50, i+offset, IF(other<15, 1, 2))),
        tbl, IF(COLUMNS(list)<2, ASTMtable, list),
        g, FILTER(tbl, CHOOSECOLS(tbl, 1)=key, ""),
        IF(isOrg, IF(key="OCH", "OH", "OL"), INDEX(g, j, 1))&" - "&INDEX(g, j, 2)
    ));

/** Fine-grained soil symbol from the plasticity chart (CL, CH, ML, MH, OL, OH, CL & ML). Inputs: LL, PL in %, optional organic (0 = inorganic). */
classify_Plasticity = LAMBDA(LL, PL, [organic],
    LET(
        organic_, IF(ISOMITTED(organic), 0, organic),
        PI, LL-PL,
        isAboveUline, IFS(LL<7, PI>LL, AND(7<=LL, LL<=15), PI>7, TRUE, PI>0.9*(LL-8)),
        isAboveAline, IF(LL<25.48, PI>4, PI>0.73*(LL-20)),
        IFS(LL<PL, "Error: Liquid limit must be larger or equal to plastic limit", isAboveUline, "Error: Above U line in plasticity chart", AND(4<=PI, PI<=7, LL>PI, PI>=0.73*(LL-20)), "CL & ML", isAboveAline, IF(organic=0, "C", "O")&IF(LL>50, "H", "L"), TRUE, IF(organic=0, "M", "O")&IF(LL>50, "H", "L"))
    ));

/** USCS group symbol from gradation and plasticity. Inputs: % passing No.4 and No.200, optional Cu = D60/D10, Cc = D30²/(D10·D60), LL, PL, organic. */
classify_USC = LAMBDA(pf4_, pf200_, [Cu_], [Cc_], [LL_], [PL_], [organic_],
    LET(
        organic, IF(ISOMITTED(organic_), 0, organic_),
        Cu, IF(ISOMITTED(Cu_), 0, Cu_),
        Cc, IF(ISOMITTED(Cc_), 0, Cc_),
        LL, IF(ISOMITTED(LL_), 0, LL_),
        PL, IF(ISOMITTED(PL_), 0, PL_),
        fines, classify_Plasticity(LL, PL),
        clayFines, LEFT(fines, 1)="C",
        isSand, pf4_-pf200_>=100-pf4_,
        isWellGraded, IF(isSand, AND(Cu>6, Cc>1, Cc<3), AND(Cu>4, Cc>1, Cc<3)),
        IFS(pf4_<pf200_, "Error: Percent passing No.4 must be larger than percent passing No.200", organic>90, "Pt", LEFT(fines, 3)="Err", fines, AND(0<=pf200_, pf200_<5), IF(isSand, IF(isWellGraded, "SW", "SP"), IF(isWellGraded, "GW", "GP")), AND(5<=pf200_, pf200_<12, isSand), IF(isWellGraded, IF(clayFines, "SW-SC", "SW-SM"), IF(clayFines, "SP-SC", "SP-SM")), AND(5<=pf200_, pf200_<12, NOT(isSand)), IF(isWellGraded, IF(clayFines, "GW-GC", "GW-GM"), IF(clayFines, "GP-GC", "GP-GM")), AND(12<=pf200_, pf200_<50, fines="CL & ML"), IF(isSand, "SC-SM", "GC-GM"), AND(12<=pf200_, pf200_<50), IF(clayFines, IF(isSand, "SC", "GC"), IF(isSand, "SM", "GM")), pf200_>=50, fines)
    ));

/** Square-root-of-time analysis of a consolidation test: Cv, d0, d100, drainage length and compression ratios. Inputs: time and dial-reading columns, fitting window ta–tb, initial height hi. */
consolidation_test = LAMBDA(Tarray, Xarray, ta, tb, hi,
    LET(
        lbl, {"Consolidation coefficient Cv =";"Initial height =";"Initial reading =";"Reading at consolidation beginning d0 =";"Reading at consolidation end d100 =";"Final reading =";"Average drainage length =";"Initial compression ri =";"Primary compression rp =";"Secondary compression rs ="},
        tf, FILTER(Tarray, (Tarray>ta)*(Tarray<=tb)),
        xf, FILTER(Xarray, (Tarray>ta)*(Tarray<=tb)),
        sqtf, SQRT(tf),
        a, SLOPE(xf, sqtf)/1.15,
        B, INTERCEPT(xf, sqtf),
        sTarray, SQRT(Tarray),
        Xcut, cutline(sTarray, Xarray, a, B),
        tc, TAKE(Xcut, 1, -1)^2,
        xc, TAKE(Xcut, 1, -1),
        Cv, 0.848*xc^2/tc^2,
        xe, TAKE(Xarray, -1),
        xs, INDEX(Xarray, 1),
        xp, B+(xc-B)/0.9,
        dx, xs-xe,
        ri, (xs-B)/dx,
        rp, (B-xp)/dx,
        rs, (xp-xe)/dx,
        h, (hi+xe-(B+xp)/2)/2,
        HSTACK(lbl, VSTACK(Cv, hi, xs, B, xp, xe, h, ri, rp, rs))
    ));

/** Time factor Tv from average degree of consolidation U (fraction, 0 ≤ U < 1), approximate formula. */
consolidation_Tv = LAMBDA(U, IF(U>0.6, -0.085-0.933*LOG(1-U), PI()/4*U^2));

/** Average degree of consolidation U (fraction) from time factor tv by Fourier series. case_: uniform (default), triangular_top, triangular_bottom, vshape, saw, sine, parabolic, circular. */
consolidation_U = LAMBDA(tv, [case_],
    LET(
        cases, {"uniform", "triangular_top", "triangular_bottom", "vshape", "saw", "sine", "parabolic", "circular"},
        caseIndex, IF(ISOMITTED(case_), 1, XMATCH(LOWER(case_), cases, 0)),
        n, SEQUENCE(1, MIN(1000, 2+INT(SQRT(160/MAX(MIN(tv), 1E-6))/PI()))),
        np, n*PI(),
        sn, IF(MOD(n, 2), (-1)^((n-1)/2), 0),
        en, IF(np^2*tv/4>700, 0, 1/EXP(np^2*tv/4)),
        cm, SWITCH(caseIndex, 1, 8*sn^4/np^2, 2, IF(MOD(n, 2), 8/np^2, 0), 3, IF(MOD(n, 2), 8/np^2, 0), 4, 16*sn^3*(np*sn-2)/np^3, 5, 32*sn^3/np^3, 6, IF(n=1, 1, 0), 7, IF(MOD(n, 2), 96/np^4, 0), 8, IF(MOD(n, 2), -16*(-1)^((n+1)/2)*BESSELJ(np/2, 1)/np^2, 0), 0, 0),
        1-BYROW(cm*en, LAMBDA(x, SUM(x)))
    ));

/** Excess pore pressure ratio u/u0 along normalized depth at time factor tv. Inputs: Xarray (z/H column), tv, case_ (same cases as consolidation_U). */
consolidation_Ux = LAMBDA(Xarray, tv, [case_],
    LET(
        cases, {"uniform", "triangular_top", "triangular_bottom", "vshape", "saw", "sine", "parabolic", "circular"},
        caseIndex, IF(ISOMITTED(case_), 1, XMATCH(LOWER(case_), cases, 0)),
        n, SEQUENCE(1, MIN(1000, 2+INT(SQRT(160/MAX(MIN(tv), 1E-6))/PI()))),
        np, n*PI(),
        sn, IF(MOD(n, 2), (-1)^((n-1)/2), 0),
        en, IF(np^2*tv/4>700, 0, 1/EXP(np^2*tv/4)),
        cm, SWITCH(caseIndex, 1, 4*sn^2/np, 2, -2*(-1)^n/np, 3, 2/np, 4, 4*sn*(np*sn-2)/np^2, 5, 8*sn/np^2, 6, IF(n=1, 1, 0), 7, IF(MOD(n, 2), -8*(-1)^n/np^3, 0), 8, 2*sn/n*BESSELJ(np/2, 1), 0, 0),
        BYROW(Xarray, LAMBDA(x, SUM(cm*en*SIN(np*x))))
    ));

/** Average degree of consolidation U (fraction) from tv, closed-form approximation. */
consolidation_U_approximate = LAMBDA(tv,
    LET(
        U, 2*SQRT(tv/PI()),
        IF(U<0.6, U, 1-8/PI()^2/EXP(PI()^2/4*tv))
    ));

/** Average consolidation U for a load applied linearly over construction time factor tc. Inputs: tv, tc. */
consolidation_U_variable = LAMBDA(tv, tc, IF(tv<tc, consolidation_U(tv/2)*tv/tc, consolidation_U(tv-tc/2)));

/** Cosine of an angle in degrees. */
cosd = LAMBDA(X, COS(RADIANS(X)));

/** Coulomb trial wedge with cohesion: {W; R; P; Cs; Cw; z0; K}. Inputs: t wedge angle (radians), phi, alpha wall, beta slope, delta (deg), g unit weight, h wall height, optional coh_. */
Coulomb_cohesiveforces = LAMBDA(t, phi, alpha, beta, delta, g, h, [coh_],
    LET(
        f, RADIANS(phi),
        ar, RADIANS(alpha),
        br, RADIANS(beta),
        dr, RADIANS(delta),
        BC, h*SIN(br-ar)/SIN(br-t)/SIN(ar),
        AB, h*SQRT(1+COT(ar)^2),
        Ka, (1-SIN(f))/(1+SIN(f)),
        coh, IF(ISOMITTED(coh_), 0, coh_),
        z0, 2*coh/g/SQRT(Ka),
        Cw, coh*MAX(0, AB-z0),
        Cs, coh*MAX(0, BC-z0),
        W, g*AB*BC*SIN(ar-t)/2,
        den, SIN(f+ar+dr-t),
        rValue, (W*SIN(ar+dr)-Cw*COS(dr)-Cs*COS(ar+dr-t))/den,
        p, (W*SIN(t-f)-Cs*COS(f)-Cw*COS(ar+f-t))/den,
        K, p/(g*h^2/2),
        VSTACK(W, rValue, p, Cs, Cw, z0, K)
    ));

/** Coulomb trial wedge forces {W; R; P; K} for slip angle t (radians). Inputs: phi, alpha wall (90 = vertical), beta slope, delta (deg), g, h. */
Coulomb_forces = LAMBDA(t, phi, alpha, beta, delta, g, h,
    LET(
        f, RADIANS(phi),
        ar, RADIANS(alpha),
        br, RADIANS(beta),
        dr, RADIANS(delta),
        BC, h*SIN(br-ar)/SIN(br-t)/SIN(ar),
        AB, h*SQRT(1+COT(ar)^2),
        W, g*AB*BC*SIN(ar-t)/2,
        den, SIN(t-f-ar-dr),
        rValue, -W*SIN(ar+dr)/den,
        p, W*SIN(f-t)/den,
        K, p/(g*h^2/2),
        VSTACK(W, rValue, p, K)
    ));

/** Coulomb earth pressure coefficient. Inputs: phi, alpha, beta, delta (deg); passive: omit for Ka, any value for Kp. */
coulomb_K = LAMBDA(phi, alpha, beta, delta, [passive],
    LET(
        f, RADIANS(phi),
        ar, RADIANS(alpha),
        br, RADIANS(beta),
        dr, RADIANS(delta),
        Ka, (SIN(ar-f)/SIN(ar)/(SQRT(SIN(ar+dr))+SQRT(SIN(f+dr)*SIN(f-br)/SIN(ar-br))))^2,
        Kp, (SIN(ar+f)/SIN(ar)/(SQRT(SIN(ar-dr))-SQRT(SIN(f+dr)*SIN(f+br)/SIN(ar-br))))^2,
        IF(ISOMITTED(passive), Ka, Kp)
    ));

/** Critical Coulomb wedge by golden-section search: {slip angle (deg); W; R; P; K}. Inputs: a_, b_ (any values on first call), phi, alpha, beta, delta, g, h. */
Coulomb_maximum = LAMBDA(a_, b_, phi, alpha, beta, delta, g, h, [iter_],
    LET(
        niter, 10,
        gr, (SQRT(5)+1)/2,
        iter, IF(ISOMITTED(iter_), 1, iter_),
        A, IF(iter=1, PI()/2, a_),
        B, IF(iter=1, RADIANS(beta)+0.01, b_),
        cValue, B-(B-A)/gr,
        d, A+(B-A)/gr,
        Yc, -INDEX(Coulomb_forces(cValue, phi, alpha, beta, delta, g, h), 3),
        yd, -INDEX(Coulomb_forces(d, phi, alpha, beta, delta, g, h), 3),
        IF(iter<niter, IF(Yc<yd, Coulomb_maximum(A, d, phi, alpha, beta, delta, g, h, iter+1), Coulomb_maximum(cValue, B, phi, alpha, beta, delta, g, h, iter+1)), VSTACK(DEGREES((A+B)/2), Coulomb_forces((A+B)/2, phi, alpha, beta, delta, g, h)))
    ));

/** Critical Coulomb slip angle (deg), closed form. Inputs: phi, alpha, beta, delta (deg). */
coulomb_theta = LAMBDA(phi, alpha, beta, delta,
    LET(
        angA, RADIANS(alpha-beta),
        angB, RADIANS(beta-phi),
        angD, RADIANS(alpha+delta+phi-beta),
        coefA, SIN(angA)*SIN(angB),
        coefB, SIN(angA-angB),
        coefC, -COS(angA)*COS(angB),
        coefD, SIN(angD),
        coefE, -COS(angD),
        quadA, coefA*coefD,
        quadB, 2*coefA*coefE,
        quadC, coefB*coefE-coefC*coefD,
        disc, quadB^2-4*quadA*quadC,
        root, (-quadB-SQRT(disc))/(2*quadA),
        beta+DEGREES(ATAN2(root, 1))
    ));

/** Intersections {x, y} of a sampled curve with the line y = slope·x + intercept. Inputs: Xarray, Yarray columns, slope, intercept. */
cutline = LAMBDA(Xarray, Yarray, slope, intercept,
    LET(
        Xa, DROP(Xarray, -1),
        xb, DROP(Xarray, 1),
        Ya, DROP(Yarray, -1),
        yb, DROP(Yarray, 1),

        z, IF(denominator<>0, (intercept-Ya+slope*Xa)/denominator, -10),
        Xcut, HSTACK(Xa+z*(xb-Xa), Ya+z*(yb-Ya)),
        FILTER(Xcut, (z>0)*(z<=1))
    ));

elastic_circular_radius = LAMBDA(s, SQRT(1/(1-s)^(2/3)-1));

/** Stresses {sx, sy, sxy} per unit vertical line load (Flamant). Inputs: x horizontal offset, y depth. Multiply by the line load. */
elastic_lineload = LAMBDA(x, y,
    LET(
        temp, 2/PI()/(x^2+y^2)^2,
        sy, temp*y^3,
        sx, temp*x^2*y,
        sxy, temp*x*y^2,
        HSTACK("Stress sx,sy,sxy =", sx, sy, sxy)
    ));

/** Stresses {sx, sy, sxy} per unit horizontal line load. Inputs: x horizontal offset, y depth. Multiply by the line load. */
elastic_lineload_horizontal = LAMBDA(x, y,
    LET(
        temp, 2/PI()/(x^2+y^2)^2,
        sy, temp*x*y^2,
        sx, temp*x^3,
        sxy, temp*x^2*y,
        HSTACK("Stress sx,sy,sxy =", sx, sy, sxy)
    ));

/** Newmark influence factor under the corner of a uniformly loaded rectangle. Inputs: m = B/z, n = L/z. */
elastic_Newmark_factor = LAMBDA(m, n,
    LET(
        v, 1+m^2+n^2,
        u, ATAN(m*n/SQRT(v)),
        1/(2*PI())*(m*n*(1+v)/SQRT(v)/(1+m^2)/(1+n^2)+u)
    ));

/** Vertical stress factor at any point (x_, y_, z_) under a loaded rectangle by superposition of corner factors. pt: 4 corners, opposite corners in rows 1 and 3. */
elastic_Newmark_stress = LAMBDA(x_, y_, z_, pt,
    LET(
        z, IF(z_=0, 1, z_),
        x, x_/z,
        y, y_/z,
        xa, INDEX(pt, 1, 1)/z,
        xb, INDEX(pt, 3, 1)/z,
        ya, INDEX(pt, 1, 2)/z,
        yb, INDEX(pt, 3, 2)/z,
        Ibb, SIGN((x-xb)*(y-yb))*elastic_Newmark_factor(ABS(x-xb), ABS(y-yb)),
        Iaa, SIGN((x-xa)*(y-ya))*elastic_Newmark_factor(ABS(x-xa), ABS(y-ya)),
        Iba, SIGN((xb-x)*(y-ya))*elastic_Newmark_factor(ABS(xb-x), ABS(y-ya)),
        Iab, SIGN((xa-x)*(y-yb))*elastic_Newmark_factor(ABS(xa-x), ABS(y-yb)),
        Iaa+Ibb+Iab+Iba
    ));

/** Stresses {sz, sr, st, srz} per unit point load (Boussinesq). Inputs: rValue radial offset, z depth, optional nu_ (default 0.3). */
elastic_pointload = LAMBDA(rValue, z, [nu_],
    LET(
        nu, IF(ISOMITTED(nu_), 0.3, nu_),
        D, rValue^2+z^2,
        sz, 3/(2*PI()*z^2)/(1+(rValue/z)^2)^2.5,
        sr, PI()/2*(3*rValue^2*z/D^2.5-(1-2*nu)/(D+z*SQRT(D))),
        st, -1/2/PI()*(1-2*nu)*(z/D^1.5-1/(rValue^2+z^2+z*SQRT(D))),
        srz, 3/2/PI()*rValue*z^2/D^2.5,
        HSTACK("Stress sz,sr,st,srz =", sz, sr, st, srz)
    ));

/** Vertical stress factor under a point (x, y) at depth z for a uniformly loaded polygon, by integration over 1° sectors. pts: polygon vertices (two columns). */
elastic_sector = LAMBDA(x, y, z, pts,
    LET(
        n, 360,
        t, RADIANS(SEQUENCE(n+1, 1, 0, 1)),
        pt, HSTACK(x, y),
        ptz, TOROW((pts-pt)/z),
        txy, HSTACK(t, IF(SEQUENCE(n+1), ptz)),
        s, BYROW(txy, LAMBDA(x, elastic_sector_row(x))),
        SUM(s)/n
    ));

/** Helper for elastic_sector: contribution of one ray. Input: row with angle (radians) followed by flattened polygon coordinates. */
elastic_sector_row = LAMBDA(txy,
    LET(
        t, INDEX(txy, 1),
        V, DROP(txy, , 1),
        pts, WRAPROWS(V, 2),
        rs, intersection_polygon_ray(pts, {0, 0}, t),
        s, 1-1/(1+rs^2)^1.5,
        IFERROR(IF(COUNT(s)=2, ABS(INDEX(s, 2)-INDEX(s, 1)), s), 0)
    ));

/** Vertical stress contribution of an annular sector between radii ra and rb at depth z. dt: angular weight (sector angle / 2π). */
elastic_sector_slice = LAMBDA(ra, rb, z, dt, dt*(1/(1+ra^2/z^2)^(3/2)-1/(1+rb^2/z^2)^(3/2)));

/** Stress ratios {sy, sx, sxy} under a uniform strip load. Inputs: B full width, x offset from centre, y depth. Multiply by the pressure. */
elastic_stripload = LAMBDA(B, x, y,
    LET(
        tm, ATAN2(y, x-B/2),
        tp, ATAN2(y, x+B/2),
        sy, 1/PI()*(tp-tm+SIN(tp-tm)*COS(tp+tm)),
        sx, 1/PI()*(tp-tm-SIN(tp-tm)*COS(tp+tm)),
        sxy, 1/PI()*(SIN(tp-tm)*SIN(tp+tm)),
        HSTACK("Stress sy,sx and sxy", sy, sx, sxy)
    ));

/** Stress ratios {sy, sx, sxy} under a triangular strip load. Inputs: B loaded width, x offset, y depth. Multiply by the maximum pressure. */
elastic_triangle = LAMBDA(B, x, y,
    LET(
        a, ATAN(x/y)+ATAN((B-x)/y),
        rValue, x^2+y^2,
        D, rValue+B^2-2*B*x,
        sy, (x*a+y*(1-(rValue-B*x)/D))/B/PI(),
        sx, y*(LN(D/rValue)+rValue/D-1+x/y*a-B*x/D)/B/PI(),
        sxy, (y^2/D-y/B*a)/PI(),
        HSTACK("Stress sy, sx, sxy =", sy, sx, sxy)
    ));

/** Interpolate y at X from (Xa, Ya); X may be a column. Xa may be increasing or decreasing. mode: "linear" (default), "logx", "logy", "logxy". outside: omitted = #N/A, "extrapolate", "clamp". */
Interpolate = LAMBDA(X, Xa, Ya, [mode], [outside],
    LET(
        md, IF(ISOMITTED(mode), "LINEAR", UPPER(mode)),
        od, IF(ISOMITTED(outside), "ERROR", UPPER(outside)),
        lx, OR(md = "LOGX", md = "LOGXY"),
        ly, OR(md = "LOGY", md = "LOGXY"),
        xs, SORTBY(TOCOL(Xa), TOCOL(Xa)),
        ys, SORTBY(TOCOL(Ya), TOCOL(Xa)),
        n, ROWS(xs),
        xlo, INDEX(xs, 1),
        xhi, INDEX(xs, n),
        q, IF(od = "CLAMP", IF(X < xlo, xlo, IF(X > xhi, xhi, X)), X),
        iRaw, IFERROR(XMATCH(q, xs, -1, 2), 1),
        i, IF(iRaw > n - 1, n - 1, iRaw),
        xA, INDEX(xs, i),
        xB, INDEX(xs, i + 1),
        yA, INDEX(ys, i),
        yB, INDEX(ys, i + 1),
        t, IF(lx, LN(q/xA)/LN(xB/xA), (q - xA)/(xB - xA)),
        y, IF(ly, yA*(yB/yA)^t, yA + t*(yB - yA)),
        outRange, (q < xlo) + (q > xhi),
        IF(outRange*(od = "ERROR"), NA(), IF(xB = xA, yA, y))
    ));

/** Intersections {x, y} of a polygon with the infinite line through pt at angle t (radians). Returns {0, 0} if none. */
intersection_polygon_line = LAMBDA(polygon, pt, t,
    LET(
        X_, CHOOSECOLS(polygon, 1),
        Y_, CHOOSECOLS(polygon, 2),
        n, COUNT(X_),
        closed, AND(INDEX(X_, 1)=INDEX(X_, n), INDEX(Y_, 1)=INDEX(Y_, n)),
        X, IF(closed, X_, VSTACK(X_, INDEX(X_, 1))),
        Y, IF(closed, Y_, VSTACK(Y_, INDEX(Y_, 1))),
        xp, INDEX(pt, 1),
        yp, INDEX(pt, 2),
        Xa, DROP(X, -1),
        Xb, DROP(X, 1),
        dX, Xb-Xa,
        Ya, DROP(Y, -1),
        Yb, DROP(Y, 1),
        dY, Yb-Ya,
        det, SIN(t)*dX-COS(t)*dY,
        a, IF(det<>0, (SIN(t)*(Xb-xp)+COS(t)*(yp-Yb))/det, 0),
        b, IF(det<>0, (a>=0)*(a<=1)*(-Xa*Yb+Ya*Xb+xp*dY-yp*dX)/det, 0),
        bu, UNIQUE(FILTER(b, b<>0)),
        IFERROR(HSTACK(xp+bu*COS(t), yp+bu*SIN(t)), HSTACK(0, 0))
    ));

/** Sorted distances from pt to the polygon boundary along a ray at angle t (radians). */
intersection_polygon_ray = LAMBDA(polygon, pt, t,
    LET(
        XY, intersection_polygon_line(polygon, pt, t),
        X, CHOOSECOLS(XY, 1)-INDEX(pt, 1),
        Y, CHOOSECOLS(XY, 2)-INDEX(pt, 2),
        proj, X*COS(t)+Y*SIN(t),
        rValue, ROUND(SQRT(X^2+Y^2), 8)*(proj>0),
        SORT(UNIQUE(FILTER(rValue, rValue<>0)))
    ));

minimum = LAMBDA(A, B, [iter_],

    ));

/** Alias of polygon_area_centroid: centroid and area of a polygon. */
polygon_centroidArea = LAMBDA(polygon, polygon_area_centroid(polygon));

/** Weight and moment {W, M} of a polygonal block about point Pt. Inputs: XY vertices, g unit weight, Pt {x, y}. Per unit thickness. */
polygon_weightMoment = LAMBDA(XY, g, Pt,
    LET(
        CA, polygon_centroidArea(XY),
        A, INDEX(CA, 2, 2),
        Xc, INDEX(CA, 1, 2),

/** Resultant force and moment {F, M} of a piecewise-linear pressure profile about depth d0. Inputs: d depth column, p pressure column, d0. */
pressure_forceMoment = LAMBDA(d, p, d0,
    LET(
        n, COUNT(d),
        dt, DROP(d, 1),
        db, DROP(d, -1),
        Pt, DROP(p, 1),
        pb, DROP(p, -1),
        force, SUMPRODUCT(Pt+pb, dt-db)/2,
        moment, SUMPRODUCT(dt-db, db*(2*pb+Pt)+dt*(2*Pt+pb)-3*d0*(Pt+pb))/6,
        HSTACK(force, moment)
    ));

/** Rankine active coefficient for a sloping backfill. Inputs: phi, beta (deg). */
Rankine_K = LAMBDA(phi, beta,
    LET(
        f, RADIANS(phi),
        b, RADIANS(beta),
        (COS(b)-SQRT(COS(b)^2-COS(f)^2))/(COS(b)+SQRT(COS(b)^2-COS(f)^2))
    ));

/** Rankine stresses on a wall from the Mohr circle (labelled table). Inputs: phi, coh, alpha wall angle, beta slope (deg), gz vertical stress, passive (omit = active). */
Rankine_stress = LAMBDA(phi, coh, alpha, beta, gz, [passive],
    LET(
        f, RADIANS(phi),
        br, RADIANS(beta),
        sa, gz*COS(br)^2,

        sm, INDEX(smf, 1, 2),

        tb, INDEX(st, 2),
        ib, DEGREES(ATAN2(sb, tb)),
        label, {"Lateral pressure coefficient and inclination =";"Normal and shear stress parallel to slope =";"Normal and shear stress at pole =";"Normal and shear stress on wall ="},
        sta, HSTACK(sa, ta),

/** Burland–Burbidge settlement from SPT. Inputs: z, N columns, q pressure, B width, optional LB = L/B, D_ depth, H layer thickness, sc preconsolidation pressure. */

/** Resultant force, moment and point of application of normal/shear stresses varying linearly along segment AB. Inputs: A, B {x, y}; sa, sb {normal, shear}. */
stress_ForceMoment = LAMBDA(A, B, sa, sb,
    LET(
        v, B-A,
        AB, SQRT(INDEX(v, 1)^2+INDEX(v, 2)^2),
        sm, (sa+sb)/2,
        fx, -(INDEX(sm, 1)*INDEX(v, 2)-INDEX(sm, 2)*INDEX(v, 1)),
        fy, (INDEX(sm, 1)*INDEX(v, 1)+INDEX(sm, 2)*INDEX(v, 2)),
        moment, (INDEX(sa, 1)+2*INDEX(sb, 1))*AB^2/6,
        zP, (INDEX(sa, 1)+2*INDEX(sb, 1))/INDEX(sm, 1)/6,
        p, A*(1-zP)+zP*B,
        VSTACK(HSTACK("Force =", fx, fy), HSTACK("Moment =", moment, ""), HSTACK("Point of application =", p))
    ));

/** Normal and shear stress {s, t} on a plane at angle alpha (deg) using the Mohr circle pole. Inputs: sa, ta pole stresses, alpha, sm circle centre. */
Stress_Mohr = LAMBDA(sa, ta, alpha, sm,
    LET(
        ar, RADIANS(alpha),
        am, 2*(sm-sa)*COS(ar)-2*ta*SIN(ar),
        s, sa+am*COS(ar),
        t, ta+am*SIN(ar),
        HSTACK(s, t)
    ));

/** Mohr–Coulomb failure circle through a known stress (sa, ta) on plane A (deg): centre, radius, pole, principal and failure stresses. passive: omit = active. */
stress_Mohr_failure = LAMBDA(sa, ta, A, phi, coh, [passive],
    LET(
        f, RADIANS(phi),
        sgn, IF(ISOMITTED(passive), -1, 1),
        sm, sa/COS(f)^2+coh*TAN(f)+sgn*SQRT((coh+sa*TAN(f))^2-ta^2)/COS(f),
        rValue, SQRT((sa-sm)^2+ta^2),
        ar, IF(A=0, 0.000001, RADIANS(A)),
        am, 2*(sm-sa)*COS(ar)-2*ta*SIN(ar),
        sp, sa+am*COS(ar),
        tp, ta+am*SIN(ar),
        s1_, sm+rValue,
        s3_, sm-rValue,
        a1_, DEGREES(ATAN2(s1_-sp, -tp)),
        a3_, DEGREES(ATAN2(s3_-sp, -tp)),
        ap, DEGREES(ATAN2(sm-sp, -tp))+90,
        sf, sm*COS(f)^2-coh*SIN(2*f)/2,
        tfp, coh*COS(f)^2+sm*SIN(f)*COS(f),
        afp, DEGREES(ATAN2(sf-sp, tfp-tp)),
        tfn, -tfp,
        afn, DEGREES(ATAN2(sf-sp, tfn-tp)),
        header, {"", "Normal stress", "Shear stress", "Surface orientation (deg)"},
        ra, HSTACK("Center =", sm, "", ""),
        rb, HSTACK("Radius =", rValue, "", ""),
        rj, HSTACK(header),
        rd, HSTACK("Pole", sp, tp, ap),
        re, HSTACK("Major principal stress", s1_, 0, a1_),
        rf, HSTACK("Minor principal stress", s3_, 0, a3_),
        rg, HSTACK("Minor principal stress", s3_, 0, a3_),
        rh, HSTACK("Failure stress", sf, tfp, afp),
        ri, HSTACK("Failure stress", sf, tfn, afn),
        VSTACK(ra, rb, rj, rd, re, rf, rg, rh, ri)
    ));

/** Mohr circle and pole from stresses on two perpendicular planes: centre, radius, pole, principal stresses. Inputs: sa, ta on plane A (deg), sb normal stress on the perpendicular plane. */
stress_Mohr_pole = LAMBDA(sa, ta, A, sb,
    LET(
        tb, -ta,
        B, A+90,
        sm, (sa+sb)/2,
        rValue, SQRT((sa-sb)^2/4+ta^2),
        ar, IF(A=0, 0.000001, RADIANS(A)),
        am, 2*(sm-sa)*COS(ar)-2*ta*SIN(ar),
        sp, sa+am*COS(ar),
        tp, ta+am*SIN(ar),
        s1_, sm+rValue,
        s3_, sm-rValue,
        a1_, DEGREES(ATAN2(s1_-sp, -tp)),
        a3_, DEGREES(ATAN2(s3_-sp, -tp)),
        ap, DEGREES(ATAN2(sm-sp, -tp))+90,
        header, {"", "Normal stress", "Shear stress", "Surface orientation (deg)"},
        VSTACK(HSTACK("Center =", sm, "", ""), HSTACK("Radius =", rValue, "", ""), HSTACK(header), HSTACK("Stress b", sb, tb, B), HSTACK("Pole", sp, tp, ap), HSTACK("Major principal stress", s1_, 0, a1_), HSTACK("Minor principal stress", s3_, 0, a3_))
    ));

        header, {"SBT thickness", "SBT", "SBTIc thickness", "SBTIc"},
        hSBT, CHOOSECOLS(dSBT, 2),
        hSBTn, CHOOSECOLS(dSBTn, 2),
        VSTACK(header, HSTACK(h, hSBT, hn, hSBTn))
    ));

/** Robertson soil behaviour type for a CPT profile: {Qn, Fr, SBT, name, Ic, SBT(Ic), name}. Inputs: dqs (depth, qc, fs), pa, n exponent, g, gw, dw. */
CPT_Robertson_SBT = LAMBDA(dqs, pa, n, g, gw, dw,
    LET(
        name, {"Sensitive fine-grained", "Clay - organic soil", "Clays: clay to silty clay", "Silt mixtures: clayey silt & silty clay", "Sand mixtures: silty sand to sandy silt", "Sands: clean sands to silty sands", "Dense sand to gravelly sand", "Stiff sand to clayey sand", "Stiff fine-grained", "NA"},
        d, CHOOSECOLS(dqs, 1),
        Q, CHOOSECOLS(dqs, 2),
        F, CHOOSECOLS(dqs, 3),
        sigt, g*d,
        sige, sigt-gw*IF(d>dw, d-dw, 0),
        Qn, (Q-sigt)/pa*(pa/sige)^n,
        Fr, F/(Q-sigt)*100,
        QR, HSTACK(Qn, Fr),
        SBT, BYROW(QR, LAMBDA(A, CPT_Robertson_SBT_row(CHOOSECOLS(A, 1), CHOOSECOLS(A, 2)))),
        SBTname, INDEX(name, 1, SBT),
        Ic, SQRT((3.47-LOG(ABS(Qn)))^2+(1.22+IF(Fr>0, LOG(Fr), -4))^2),
        SBTIc, ROUND(9.2-1.877*Ic, 0),
        SBTnameIc, INDEX(name, 1, SBTIc),
        HSTACK(QR, SBT, SBTname, Ic, SBTIc, SBTnameIc)
    ));

/** Helper for CPT_Robertson_SBT: SBT zone (1–9, 0 = outside chart) for one point. Inputs: Qt_, Fr_ (%). */
CPT_Robertson_SBT_row = LAMBDA([Qt_], [Fr_],
    LET(
        Qt, IF(OR(ISOMITTED(Qt_), ISBLANK(Qt_)), -1, Qt_),
        Fr, IF(OR(ISOMITTED(Fr_), ISBLANK(Fr_)), -1, Fr_),
        x, IF(Fr>0, LOG(Fr), -2),
        LQ, IF(Qt>0, LOG(Qt), -2),
        a, -0.6662*x^3-1.7072*x^2-1.5868*x+0.4627,
        b, 1.2802*x^3-1.2855*x^2+1.0234*x-0.1389,
        cValue, 1.1037*x^3-0.009*x^2+0.6298*x+0.7673,
        d, 1.0659*x^3-0.0174*x^2-2.6277*x+3.3431,
        e, 0.7223*x^4+0.6844*x^3+0.4039*x^2+0.6797*x+1.2115,
        f, 0.2897*x^3+0.8671*x^2+0.9718*x+1.8201,
        g, 1.8669*x^4+4.5888*x^3+4.4977*x^2+2.5417*x+2.9667,
        h, 9.682*x-3.7227,
        IFS(OR(-1>x, x>1, Qt<0, LQ>3), 0, LQ<=a, 1, LQ<=b, 2, LQ>g, 7, AND(LQ>d, LQ>h), 8, AND(LQ>d, LQ<=h), 9, LQ<=cValue, 3, LQ<=e, 4, LQ<=f, 5, TRUE, 6)
    ));

/** Schmertmann settlement from a CPT profile (total settlement + layer table). Inputs: zqc (depth, qc), k = E/qc, B, D, L, qn net pressure, g unit weight. */
CPT_settlement = LAMBDA(zqc, k, B, D, L, qn, g,
    LET(
        z, CHOOSECOLS(zqc, 1),
        qc, CHOOSECOLS(zqc, 2),
        za, (DROP(z, 1)+DROP(z, -1))/2,
        qca, (DROP(qc, 1)+DROP(qc, -1))/2,
        Iz, CPT_settlement_Iz(za, B, D, L, qn, g),
        cValue, MAX(1-g*D/qn/2, 0.5),
        dz, DROP(z, 1)-DROP(z, -1),
        E, k*qca,
        eps, IF(E>0, cValue*Iz*qn/E, 0),
        s, SUM(eps*dz),
        zE, FILTER(HSTACK(za, qca, E, Iz, eps), Iz>0),
        label, {"depth", "qc", "E", "I", "eps"},
        VSTACK(HSTACK("Settlement =", s, "", "", ""), label, zE)
    ));

/** Schmertmann strain influence factor Iz at depth z (accepts a column). Inputs: z, B, D, L, qn net pressure, g unit weight. */
CPT_settlement_Iz = LAMBDA(z, B, D, L, qn, g,
    LET(
        x, 1-B/L/2,
        zm, D+B*x,
        zl, D+4*B*x,
        ID, 0.2*x,
        sm, g*zm,
        Im, 0.5+0.1*SQRT(qn/sm),
        (D<=z)*(z<zm)*(ID+(Im-ID)*(z-D)/(zm-D))+(zm<=z)*(z<=zl)*(Im-Im*(z-zm)/(zl-zm))
    ));

/** Perpendicular distances from point M {x, y} to the line through each segment of polyline pl. */
distance_point_polyline = LAMBDA(M, pl,
    LET(
        A, DROP(pl, -1),
        B, DROP(pl, 1),
        AB, CHOOSECOLS(B-A, 1)^2+CHOOSECOLS(B-A, 2)^2,
        AM, CHOOSECOLS(A-M, 1)^2+CHOOSECOLS(A-M, 2)^2,
        ABAM, CHOOSECOLS(A-M, 1)*CHOOSECOLS(B-A, 1)+CHOOSECOLS(A-M, 2)*CHOOSECOLS(B-A, 2),
        SQRT(AM-ABAM^2/AB)
    ));

/** Bilinear interpolation in a 2-D table. Inputs: x, Y query; XY table with x values in row 1 and y values in column 1 (ascending). */
interpolate_table = LAMBDA(x, Y, XY,
    LET(
        Xa, DROP(CHOOSEROWS(XY, 1), , 1),
        Ya, DROP(CHOOSECOLS(XY, 1), 1),
        Za, DROP(XY, 1, 1),
        cValue, XMATCH(x, Xa, -1),
        rValue, XMATCH(Y, Ya, -1),
        Xl, INDEX(Xa, cValue),
        Xu, INDEX(Xa, cValue+1),
        Yl, INDEX(Ya, rValue),
        Yu, INDEX(Ya, rValue+1),
        a, (Xu-Xl)*(Yu-Yl),
        Nll, (Xu-x)*(Yu-Y),
        Nul, (x-Xl)*(Yu-Y),
        Nuu, (x-Xl)*(Y-Yl),
        Nlu, (Xu-x)*(Y-Yl),
        Zll, INDEX(Za, rValue, cValue),
        Zlu, INDEX(Za, rValue+1, cValue),
        Zuu, INDEX(Za, rValue+1, cValue+1),
        Zul, INDEX(Za, rValue+1, cValue),
        (Nll*Zll+Nlu*Zlu+Nuu*Zuu+Nul*Zul)/a
    ));

/** Intersection points {x, y} of a polyline with a circle. Inputs: pline vertices, center {x, y}, radius. */
intersection_polyline_circle = LAMBDA(pline, center, radius,
    LET(
        pl, pline-center,
        A, DROP(pl, -1),
        B, DROP(pl, 1),
        AB, B-A,
        OAs, CHOOSECOLS(A, 1)^2+CHOOSECOLS(A, 2)^2,
        ABs, CHOOSECOLS(AB, 1)^2+CHOOSECOLS(AB, 2)^2,
        AAB, CHOOSECOLS(A, 1)*CHOOSECOLS(AB, 1)+CHOOSECOLS(A, 2)*CHOOSECOLS(AB, 2),
        discri, AAB^2-ABs*(OAs-radius^2),
        ap, IF(discri>=0, (-AAB-SQRT(discri))/ABs, -1),
        bp, IF(discri>=0, (-AAB+SQRT(discri))/ABs, -1),
        X, VSTACK(HSTACK(A, AB, ap), HSTACK(A, AB, bp)),
        Y, FILTER(X, (0<=CHOOSECOLS(X, 5))*(CHOOSECOLS(X, 5)<=1)),
        CHOOSECOLS(Y, {1, 2})+CHOOSECOLS(Y, 5)*CHOOSECOLS(Y, {3, 4})+center
    ));

/** Intersections of polyline pl with polyline ql: {parameter, x, y}. Optional m keeps the m points with the largest x. */
intersection_polyline_polyline = LAMBDA(pl, ql, [m],
    LET(
        mm, IF(ISOMITTED(m), 0, m),
        n, ROWS(pl)-1,
        qls, IF(SEQUENCE(n), TOROW(ql)),
        Xa, DROP(pl, -1),
        Xb, DROP(pl, 1),
        xpls, HSTACK(Xa, Xb, qls),
        a, BYROW(xpls, LAMBDA(row_, INDEX(intersection_segment_polyline(row_), 1, 1))),
        X, HSTACK(a, Xa+a*(Xb-Xa)),
        Y, FILTER(X, CHOOSECOLS(X, 1)<>0),
        IF(mm>0, TAKE(SORT(Y, 2, -1), mm), Y)
    ));

/** Helper for intersection_polyline_polyline: segment parameter of the intersection between one segment and a polyline. Input: flattened row (segment, then polyline). */
intersection_segment_polyline = LAMBDA(row,
    LET(
        pl, WRAPROWS(row, 2),
        pline, DROP(pl, 2),
        ptA, DROP(pline, -1),
        ptB, DROP(pline, 1),
        AB, ptB-ptA,
        ptC, CHOOSEROWS(pl, 1),
        ptD, CHOOSEROWS(pl, 2),
        CD, ptD-ptC,
        AC, ptC-ptA,
        det, INDEX(CD, 1)*CHOOSECOLS(AB, 2)-INDEX(CD, 2)*CHOOSECOLS(AB, 1),
        a, IF(det<>0, (INDEX(CD, 1)*CHOOSECOLS(AC, 2)-INDEX(CD, 2)*CHOOSECOLS(AC, 1))/det, 0),
        b, IF(det<>0, (a>=0)*(a<=1)*(CHOOSECOLS(AB, 1)*CHOOSECOLS(AC, 2)-CHOOSECOLS(AB, 2)*CHOOSECOLS(AC, 1))/det, 0),
        a_b, HSTACK(a, b),
        fab, IFERROR(FILTER(a_b, (CHOOSECOLS(a_b, 1)>=0)*(CHOOSECOLS(a_b, 1)<=1)*(CHOOSECOLS(a_b, 2)>=0)*(CHOOSECOLS(a_b, 2)<=1)), {0, 0}),
        CHOOSECOLS(fab, 2)
    ));

/** Intersection {x, y} of a log spiral (centre pt, radius rValue, angle phi) with segment AB_ by Newton iteration. */
intersection_spiral_segment = LAMBDA(AB_, pt, rValue, phi, [x_], [iter_],
    LET(
        niter, 10,
        iter, IF(ISOMITTED(iter_), 1, iter_),
        ab, AB_-pt,
        X, IF(ISOMITTED(x_), INDEX(ab, 1, 1), x_),
        d, TAN(RADIANS(phi)),
        S, (INDEX(ab, 2, 2)-INDEX(ab, 1, 2))/(INDEX(ab, 2, 1)-INDEX(ab, 1, 1)),
        Y, INDEX(ab, 1, 2)+S*(X-INDEX(ab, 1, 1)),
        f, LN(X^2+Y^2)-2*LN(rValue)+2*d*ATAN(Y/X),
        df, 2*(X*(1+d*S)+Y*(S-d))/(X^2+Y^2),
        xn, X-f/df,
        yn, INDEX(ab, 1, 2)+S*(xn-INDEX(ab, 1, 1)),
        IF(iter>niter, pt+HSTACK(xn, yn), intersection_spiral_segment(AB_, pt, rValue, phi, xn, iter+1))
    ));

/** Broms lateral capacity of a short pile in cohesionless soil: {V; depth of max moment; Mmax}. Inputs: D embedment, phi, g, B diameter, e load eccentricity. */
lateralpile_Broms_cohesionless = LAMBDA(D, phi, g, B, e,
    LET(
        f, RADIANS(phi),
        Kp, (1+SIN(f))/(1-SIN(f)),

        dm, SQRT(6*V/(B*g*Kp))/3,
        m, (e+2/3*dm)*V,
        VSTACK(V, dm, m)
    ));

/** Broms short pile in cohesive soil: {required D; rotation depth Dr; depth of max moment; Mmax}. Inputs: V load, e eccentricity, Su, B diameter. */

    ));

/** Plotting points of a log spiral r = rValue·exp(−θ·tanφ). Inputs: pt centre, rValue, phi (deg), optional ta_, tb_ angles (deg), n_ points (default 100). */

/** Schmertmann CPT pile capacity: ultimate, shaft, tip, average fs and qc. Inputs: table (z, qc, fs, optional soil), A tip area, p perimeter, L length, pile (steel, concrete, timber). */
pile_capacity_CPT_Schmertman = LAMBDA(z_qc_fs_soil, A, p, L, pile,

        z, CHOOSECOLS(z_qc_fs_soil, 1),
        qc, CHOOSECOLS(z_qc_fs_soil, 2),

        nz, COUNT(z),

        soil, IFERROR(XMATCH(LOWER(soil_), {"clay", "silt", "sand", "gravel"}, 1), 3),
        B, 4*A/p,
        Ksa, {0.78, 3.385, 0.1775;0.83, 1.629, 0.1736;0.98, 4.23, 0.1775},
        Ks, INDEX(Ksa, k, 1)+INDEX(Ksa, k, 2)/EXP(z/B)^INDEX(Ksa, k, 3),

        fsc, IF(k<=2, IF(fs<0.04, fs, 0.04), IF(fs<=0.03, fs, 0.03+(fs-0.03)*0.294)),
        fsx, IF(soil<=2, fsc, fsg),

        dzb, FILTER(dz, (zd>=L)*(zd<=Lb)),
        qx, SCAN(0, qbdz, LAMBDA(a, v, a+v)),
        zx, SCAN(0, dzb, LAMBDA(a, v, a+v)),

    ));

/** Shaft resistance by the α method. Inputs: z, Su columns, pa atmospheric pressure, p perimeter, L length, author (terzaghi, reese). */
pile_shaft_alpha = LAMBDA(z, Su, pa, p, L, author,
    LET(
        authors, {"terzaghi", "reese"},
        a, IFERROR(XMATCH(LOWER(author), authors, 1), 0),
        Sa, Su/pa,
        at, -0.0744*Sa^3+0.477*Sa^2-1.0243*Sa+1.0924,
        alpha, IFS(a=1, IF(at<1, at, 1), a=2, IF(Sa<=1.5, 0.55, IF(Sa<2.5, 0.55-0.1*(Sa-1.5), 0.45)), TRUE, 0),
        fn, (DROP(alpha*Su, 1)+DROP(alpha*Su, -1))/2,
        dz, DROP(z, 1)-DROP(z, -1),
        qu, p*SUM(FILTER(fn*dz, DROP(z, 1)<=L)),
        label, {"Ultimate shaft resistance ="},
        HSTACK(label, qu)
    ));

/** Adhesion factor α. Inputs: Su = Su/pa (normalized), author (terzaghi, reese). */
pile_sidefriction_alpha = LAMBDA(Su, author,
    LET(
        authors, {"terzaghi", "reese"},
        a, IFERROR(XMATCH(LOWER(author), authors, 1), 0),
        at, -0.0744*Su^3+0.477*Su^2-1.0243*Su+1.0924,
        IFS(a=1, IF(at<1, at, 1), a=2, IF(Su<=1.5, 0.55, IF(Su<2.5, 0.55-0.1*(Su-1.5), 0.45)), TRUE, 0)
    ));

/** λ coefficient for pile shaft friction (Vijayvergiya–Focht chart fit). Input: D_ pile length in m (or ft if ft is supplied). */
pile_sidefriction_lambda = LAMBDA(D_, [ft],

        0.091+23.37*(d+27.05)^(-1.227)
    ));

/** Meyerhof ultimate tip pressure. Inputs: pa atmospheric pressure, cValue cohesion, phi (deg). */
pile_tip_Meyerhof = LAMBDA(pa, cValue, phi,
    LET(
        f, RADIANS(phi),
        Nq, (1+SIN(f))/(1-SIN(f))*EXP(5*PI()/3*TAN(f)),
        ft, IF(phi=0, 0.1, f),
        Nc, MAX(9, (Nq-1)/TAN(ft)),
        label, {"Ultimate tip pressure ="},
        qu, cValue*Nc+pa*Nq*TAN(f)/2,
        HSTACK(label, qu)
    ));

/** Meyerhof pile tip factors {Nq; Nc} for phi (deg). */
pile_tip_meyerhof_factor = LAMBDA(phi,
    LET(
        f, RADIANS(phi),
        Nq, (1+SIN(f))/(1-SIN(f))*EXP(5*PI()/3*TAN(f)),
        ft, IF(phi=0, 0.1, f),
        Nc, MAX(9, (Nq-1)/TAN(ft)),
        VSTACK(Nq, Nc)
    ));

/** Vesic ultimate tip pressure with reduced rigidity index: {qu; Nq; Nc; Irr; σm}. Inputs: sv vertical stress, pa, cValue cohesion, optional phi_ (deg), Dr_ relative density. */
pile_tip_vesic = LAMBDA(sv, pa, cValue, [phi_], [Dr_],
    LET(
        phi, IF(ISOMITTED(phi_), 0, phi_),
        Dr, IF(ISOMITTED(Dr_), 0.1, Dr_),
        f, RADIANS(phi),
        K0, 1-SIN(f),
        sm, (1+2*K0)/3*sv,
        Es, 1000*Dr*pa,
        nu, 0.1+0.3*(phi-25)/20,
        Gs, Es/(1+nu)/2,
        Ir, Gs/(cValue+sm*TAN(f)),
        ev, IF(phi=0, 0, 0.005*(1-(phi-25)/20)*sv/pa),
        Irr, IF(phi=0, 305.06*cValue/pa-12.5, Ir/(1+ev*Ir)),
        Nq, IF(phi=0, 1, 3/(3-SIN(f))*EXP((PI()/2-f)*TAN(f))*TAN(PI()/4+f/2)^2*Irr^(4/3*SIN(f)/(1+SIN(f)))),
        Nc, IF(phi=0, 4/3*(LN(Irr)+1)+PI()/2+1, (Nq-1)/TAN(f)),
        qu, cValue*Nc+sm*Nq,
        label, {"Ultimate pressure =";"Nq =";"Nc =";"Reduced rigidity index Irr =";"Mean stress ="},
        HSTACK(label, VSTACK(qu, Nq, Nc, Irr, sm))
    ));

/** Vesic pile tip factors {Nq; Nc}. Inputs: phi (deg), Ir rigidity index. */
pile_tip_vesic_factor = LAMBDA(phi, Ir,
    LET(
        f, RADIANS(phi),
        Nq, 3/(3-SIN(f))*EXP((PI()/2-f)*TAN(f))*TAN(PI()/4+f/2)^2*Ir^(4/3*SIN(f)/(1+SIN(f))),
        Nc, IF(phi=0, 4/3*(LN(Ir)+1)+PI()/2+1, (Nq-1)/TAN(f)),
        VSTACK(Nq, Nc)
    ));

/** Ménard rheological factor α. Inputs: soil (peat, clay, silt, sand, gravel, rock), Epl = E/pl ratio. */
PMT_alpha = LAMBDA(soil, Epl,
    LET(
        soils, {"peat", "clay", "silt", "sand", "gravel", "rock"},
        s, IFERROR(XMATCH(LOWER(soil), soils, 1), 0),

/** Equivalent limit pressure ple (geometric mean from D to D + 1.5B) and equivalent embedment De: {ple; De}. Inputs: z, pl columns, B, D. */
PMT_bearing_limitPressure = LAMBDA(z, pl, B, D,
    LET(
        zmax, D+3*B/2,
        pld, FILTER(pl, (z>=D)*(z<=zmax), GEOMEAN(pl)),
        ple, GEOMEAN(pld),
        plu, FILTER(pl, z<=D, ple),
        De, D*IFERROR(AVERAGE(plu)/ple, 1),
        VSTACK(ple, De)
    ));

/** Ménard ultimate bearing pressure from pressuremeter data. Inputs: z, pl columns, B, D, L, soil category (I, II, III, IIIA), g unit weight. */

        ple, INDEX(pleDe, 1),
        De, INDEX(pleDe, 2),

        label, {"Ultimate pressure =";"Pressure limit ple=";"Factor k ="},
        HSTACK(label, VSTACK(qu, ple, k))
    ));

/** Ménard bearing factor k, interpolated between strip and square. Inputs: H = De/(B/2), BL = B/L, soil category (I, II, III, IIIA). */
PMT_bearing_Menard_k = LAMBDA(H, BL, soil,

        k0_+(k1_-k0_)*BL
    ));

/** Centroid and area of a polygon: {"Centroid =", xc, yc; "Area =", A, ""}. Input: vertices in two columns (first vertex may be repeated). */
polygon_area_centroid = LAMBDA(polygon,
    LET(
        closed, AND(TAKE(polygon, 1, 1)=TAKE(polygon, -1, 1), TAKE(polygon, 1, -1)=TAKE(polygon, -1, -1)),
        XY, IF(closed, polygon, VSTACK(polygon, TAKE(polygon, 1))),
        X, CHOOSECOLS(XY, 1),
        Y, CHOOSECOLS(XY, 2),
        Xb, DROP(X, -1),
        Xt, DROP(X, 1),
        Yb, DROP(Y, -1),
        Yt, DROP(Y, 1),
        Z, Xt*Yb-Xb*Yt,
        a, SUM(Z)/2,
        Xc, SUMPRODUCT(Xt+Xb, Z)/6/a,
        Yc, SUMPRODUCT(Yt+Yb, Z)/6/a,
        VSTACK(HSTACK("Centroid = ", Xc, Yc), HSTACK("Area = ", ABS(a), ""))
    ));

/** TRUE if point (x, y) lies below exactly one edge of the polygon (xa, ya) — vertical ray test. */
Polygon_inside = LAMBDA(x, y, xa, ya,
    LET(
        xd, DROP(xa, -1),
        yd, DROP(ya, -1),
        xas, DROP(xa, 1)-xd,
        yas, DROP(ya, 1)-yd,
        a, (x-xd)/xas,
        b, yd+a*yas,
        t, (a>=0)*(a<=1)*(b>=y)=1,
        SUM(N(t))=1
    ));

/** Linear interpolation of y at X along a polyline (two columns), searching for the segment starting at or below X. Used for root finding. */
polyline_subdivide = LAMBDA(X, polyline,
    LET(
        Xa, CHOOSECOLS(polyline, 1),
        Ya, CHOOSECOLS(polyline, 2),
        i, XMATCH(X, Xa, -1),
        S, (INDEX(Ya, i+1)-INDEX(Ya, i))/(INDEX(Xa, i+1)-INDEX(Xa, i)),
        INDEX(Ya, i)+S*(X-INDEX(Xa, i))
    ));

/** Schmertmann strain influence factor at depth z. Inputs: Imax, B, za footing depth, optional BL_ (0 = square, 1 = strip). */
settlement_Schmertman_factor = LAMBDA(z, Imax, B, za, [BL_],
    LET(
        BL, IF(ISOMITTED(BL_), 0, BL_),
        I, 0.2-0.1*BL,
        zb, za+B-B/2*BL,
        zc, za+4*B-2*B*BL,
        IFS(AND(za<=z, z<zb), I+(Imax-I)*(z-za)/(zb-za), AND(za<=z, z<=zc), Imax-Imax*(z-zb)/(zc-zb), TRUE, 0)
    ));

/** Factor of safety of a circular slip in undrained clay (φ = 0) using sector geometry. Inputs: pline ground surface, center, radius, cu, unit weight, detail (any value for details). */
slope_circle_cohesive = LAMBDA(pline, center, radius, cu, unitweight, [detail],
    LET(
        AB, intersection_polyline_circle(pline, center, radius),
        A, TAKE(AB, 1),
        B, TAKE(AB, -1),
        Ar, A-center,
        Br, B-center,
        ta, ATAN2(INDEX(Ar, 1), INDEX(Ar, 2)),
        tb, ATAN2(INDEX(Br, 1), INDEX(Br, 2)),
        tav, (ta+tb)/2,
        theta, ABS(ta-tb),
        arclength, radius*theta,
        resisting, cu*radius*arclength,
        area_sector, radius^2/2*theta,
        centroid_sector, 4*radius/(3*theta)*SIN(theta/2)*HSTACK(COS(tav), SIN(tav))+center,
        pts, IFERROR(FILTER(pline, CHOOSECOLS((pline-A)*(pline-B), 1)<0), A),
        exterior, VSTACK(center, SORT(UNIQUE(VSTACK(AB, pts)))),
        pgac, polygon_area_centroid(exterior),
        centroid_exterior, INDEX(pgac, 1, {2, 3}),
        area_exterior, INDEX(pgac, 2, 2),
        area, area_sector-area_exterior,
        centroid, (centroid_sector*area_sector-centroid_exterior*area_exterior)/area,
        driving, area*unitweight*INDEX(centroid-center, 1),
        FS, resisting/driving,
        details, VSTACK(HSTACK(FS, ""), HSTACK(area, arclength), centroid, A, B),
        IF(ISOMITTED(detail), FS, details)
    ));

/** Geometry of the soil mass above a circular slip: centroid, entry/exit points and angles, area. Inputs: pl ground polyline, center, radius. */
slope_circle_polyline = LAMBDA(pl, center, radius,
    LET(
        pts, DROP(circle(center, radius), 1),
        AB, intersection_polyline_polyline(pts, pl, 2),
        A, INDEX(AB, 1, {2, 3}),
        B, INDEX(AB, 2, {2, 3}),
        ta, ATAN2(INDEX(A-center, 1), INDEX(A-center, 2)),
        tb, ATAN2(INDEX(B-center, 1), INDEX(B-center, 2)),
        t, ATAN2(CHOOSECOLS(pts-center, 1), CHOOSECOLS(pts-center, 2)),
        top, IFERROR(FILTER(pl, CHOOSECOLS(pl-A, 1)*CHOOSECOLS(pl-B, 1)<0), A),
        tspg, HSTACK(t, pts),
        spg, CHOOSECOLS(FILTER(tspg, CHOOSECOLS(tspg-ta, 1)*CHOOSECOLS(tspg-tb, 1)<0), {2, 3}),
        plc, VSTACK(B, spg, A, top),
        pac, polygon_area_centroid(plc),
        centroid, INDEX(pac, 1, {2, 3}),
        area, INDEX(pac, 2, 2),
        VSTACK(HSTACK(centroid, ""), HSTACK(A, DEGREES(ta)), HSTACK(B, DEGREES(tb)), HSTACK(area, "", ""))
    ));

/** Factor of safety of a circular slip in undrained clay using polygon geometry. Inputs: pline, center, radius, cu, unit weight, detail (any value for details). */
slope_circle_polyline_cohesive = LAMBDA(pline, center, radius, cu, unitweight, [detail],
    LET(
        X, slope_circle_polyline(pline, center, radius),
        centroid, INDEX(X, 1, {1, 2}),
        A, INDEX(X, 2, {1, 2}),
        ta, RADIANS(INDEX(X, 2, 3)),
        B, INDEX(X, 3, {1, 2}),
        tb, RADIANS(INDEX(X, 3, 3)),
        area, INDEX(X, 4, 1),
        theta, ABS(ta-tb),
        arclength, radius*theta,
        resisting, cu*radius*arclength,
        driving, area*unitweight*ABS(INDEX(centroid-center, 1)),
        FS, resisting/driving,
        details, VSTACK(HSTACK(FS, ""), HSTACK(area, arclength), centroid, A, B),
        IF(ISOMITTED(detail), FS, details)
    ));

/** Friction circle method (c–φ soil): factor of safety and force polygon (labelled table). Inputs: pline, pt circle centre, rValue radius, coh, phi, g, opt_ (any value: alternative Taylor factor). */
slope_friction_circle = LAMBDA(pline, pt, rValue, coh, phi, g, [opt_],
    LET(
        f, RADIANS(phi),
        opt, IF(ISOMITTED(opt_), 0, 1),
        res, slope_circle_polyline(pline, pt, rValue),
        ptC, INDEX(res, 1, {1, 2}),
        a, INDEX(res, 2, {1, 2}),
        b, INDEX(res, 3, {1, 2}),
        area, INDEX(res, 4, 1),
        w, g*area,
        ta, ATAN2(INDEX(a-pt, 1), INDEX(a-pt, 2)),
        tb, ATAN2(INDEX(b-pt, 1), INDEX(b-pt, 2)),
        theta, ABS(ta-tb),
        alpha, ATAN2(INDEX(b-a, 1), INDEX(b-a, 2)),
        La, rValue*theta,
        Lc, SQRT(SUMSQ(b-a)),
        d, rValue*La/Lc,
        OD, INDEX(ptC-pt, 1),
        DM, d/COS(alpha)-OD*TAN(alpha),
        kr, slope_friction_circle_Taylor(theta, opt),
        FSmin, TAN(f)/TAN(ASIN(OD/kr/rValue))+0.01,
        FSf, SEQUENCE(100, 1, FSmin, 0.02),
        df, FSf-slope_friction_circle_fct(FSf, f, coh, w, kr, rValue, OD, DM, alpha, Lc),
        FS, polyline_subdivide(0, HSTACK(df, FSf)),
        fd, ATAN2(FS, TAN(f)),
        Rf, kr*rValue*ABS(SIN(fd)),
        beta, 2*ATAN((-DM+SQRT(OD^2+DM^2-Rf^2))/(OD+Rf)),
        p, w*COS(alpha)/COS(alpha-beta),
        t, w*SIN(beta)/COS(alpha-beta),
        VSTACK(HSTACK(FS, ""), HSTACK(w, OD), HSTACK(La, Lc), HSTACK(p, DEGREES(beta)), HSTACK(t, DEGREES(alpha)), HSTACK(Rf, DEGREES(theta)), ptC, a, b, HSTACK(OD, DM))
    ));

/** Helper for slope_friction_circle: cohesion factor of safety for a trial frictional FS (accepts a column of FS). */
slope_friction_circle_fct = LAMBDA(FS, f, coh, w, kr, rValue, OD, DM, alpha, Lc,
    LET(
        fd, ATAN(TAN(f)/FS),
        Rf, kr*rValue*SIN(fd),
        beta, 2*ATAN((-DM+SQRT(OD^2+DM^2-Rf^2))/(OD+Rf)),
        t, w*SIN(beta)/COS(alpha-beta),
        coh*Lc/t
    ));

/** Taylor friction-circle radius factor k for central angle t (radians). opt = 1 selects the alternative formula. */
slope_friction_circle_Taylor = LAMBDA(t, opt, IF(opt=1, SQRT(2)*(1-t^2/PI()^2)/SQRT(1+COS(t)), t/2/SIN(t/2)));

/** Residual FS − FSc for the friction circle method (same inputs as slope_friction_circle_fct). */
slope_friction_FS = LAMBDA(FS, f, coh, w, kr, rValue, a, DM, alpha, ab,
    LET(
        fd, ATAN(TAN(f)/FS),
        Rf, kr*rValue*SIN(fd),
        beta, 2*ATAN((-DM+SQRT(a^2+DM^2-Rf^2))/(a+Rf)),
        t, w*SIN(beta)/COS(alpha-beta),
        FS-ABS(coh*ab/t)
    ));

/** Log-spiral slip surface: factor of safety, centroid, weight, entry/exit points, resisting and driving moments. Inputs: pl ground polyline, pt spiral centre, rValue, phi, coh, g. */
slope_logSpiral = LAMBDA(pl, pt, rValue, phi, coh, g,
    LET(
        d, TAN(RADIANS(phi)),
        pts, logSpiral(pt, rValue, phi),
        ab, intersection_polyline_polyline(pts, pl),
        a, INDEX(ab, 1, {2, 3}),
        b, INDEX(ab, 2, {2, 3}),
        Ar, a-pt,
        Br, b-pt,
        ra, SUMSQ(Ar),
        rb, SUMSQ(Br),
        ta, ATAN2(INDEX(Ar, 1), INDEX(Ar, 2)),
        tb, ATAN2(INDEX(Br, 1), INDEX(Br, 2)),
        IAB, rValue^2/2/d*ABS(EXP(-2*ta*d)-EXP(-2*tb*d)),
        top, FILTER(pl, (CHOOSECOLS(pl, 1)-INDEX(a, 1))*(CHOOSECOLS(pl, 1)-INDEX(b, 1))<0),
        spg, FILTER(pts, (CHOOSECOLS(pts-pt, 1)^2+CHOOSECOLS(pts-pt, 2)^2-ra)*(CHOOSECOLS(pts-pt, 1)^2+CHOOSECOLS(pts-pt, 2)^2-rb)<0),
        plc, VSTACK(a, spg, b, top),
        pac, polygon_area_centroid(plc),
        Xc, INDEX(pac, 1, {2, 3}),
        dg, INDEX(Xc-pt, 1),
        w, g*INDEX(pac, 2, 2),
        FS, coh*IAB/(w*dg),
        VSTACK(HSTACK(FS, "", ""), HSTACK(Xc, w), HSTACK(a, DEGREES(ta)), HSTACK(b, DEGREES(tb)), HSTACK(coh*IAB, w*dg, ""))
    ));

/** Shear and moment from a piecewise-linear pressure profile. Columns: depth, pressure, shear, moment. k = subdivisions per segment. */
shearMoment = LAMBDA(d, p, k,
    LET(
        t, SEQUENCE(1, k, 0)/k,
        da, DROP(d, -1),
        pa, DROP(p, -1),
        x, VSTACK(TOCOL(da + (DROP(d, 1) - da)*t), TAKE(d, -1)),
        q, VSTACK(TOCOL(pa + (DROP(p, 1) - pa)*t), TAKE(p, -1)),
        h, DROP(x, 1) - DROP(x, -1),
        qa, DROP(q, -1),
        qb, DROP(q, 1),
        V, VSTACK(0, SCAN(0, h*(qa + qb)/2, LAMBDA(a, v, a + v))),
        M, VSTACK(0, SCAN(0, DROP(V, -1)*h + h^2*(2*qa + qb)/6, LAMBDA(a, v, a + v))),
        HSTACK(x, q, V, M)
    ));
